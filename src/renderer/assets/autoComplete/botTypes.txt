interface ItemQuery extends Iterable<Item> {
  /** Container slots, then the player inventory (default). */
  any(): ItemQuery;
  /** Only the top part of the open window. */
  container(): ItemQuery;
  /** Only the player inventory, whether or not a window is open. */
  inventory(): ItemQuery;
  hotbar(): ItemQuery;
  /** Container rows (0-based). With cols, only slots in both. */
  rows(...rows: number[]): ItemQuery;
  /** Container columns (0-8). */
  cols(...cols: number[]): ItemQuery;
  /** Exact window slot indexes. */
  slots(...slots: number[]): ItemQuery;
  first(): Item | null;
  all(): Item[];
  has(): boolean;
  /** Sum of item.count over all matches. */
  count(): number;
  /** Clicks the first match; returns it or null. */
  click(mouseButton?: number, mode?: number): Promise<Item | null>;
  /** Resolves with the first match once it appears, or null after `ticks` (default 100). */
  wait(ticks?: number): Promise<Item | null>;
}

type Bot =  MineflayerBot & {
  botId: number;
  join: (
    gamemode: string,
  ) => undefined | Promise<undefined> | { error: string };
  isOnline: () => boolean;
  getGamemode: () => string;
  itemInfo(item: Item | null): { name: string; info: string };
  login: (password?: string) => Promise<void>; // Only works if you call this in the start of bot

  getRank: () => string | undefined;
  transferItems: (
    mode: number | 'trade' | 'gift' | 'window' | 'tradeWithKeys',
    options: {
      username?: string;
      command?: string;
      ignore?: string[];
      filter?: string[];
      itemFilter?: (item: Item) => boolean;
      keep?: boolean; // whether pass the item if not filtered
      keys?: string[]; // How many items it should click on each key  ("<keyType>,<amount>")
      forceTradeKeys?: boolean; // Whether to force trade keys even when no other items are in player inventory
      timeout?: number | null;
      shiftClick?: boolean; // Whether to shift click items into window (windowBehaviour only)}) => void
    },
  ) => Promise<void>;

  /** Search items by name/lore/type in the open window and/or inventory. Strings: case-insensitive substring, colors and small caps stripped. Arrays match any entry. */
  items: (
    query?:
      | string
      | RegExp
      | (string | RegExp)[]
      | ((item: Item) => boolean)
      | {
          name?: string | RegExp | ((text: string) => boolean) | (string | RegExp)[];
          lore?: string | RegExp | ((text: string) => boolean) | (string | RegExp)[];
          type?: string | string[];
          where?: (item: Item) => boolean;
          caseSensitive?: boolean;
        },
  ) => ItemQuery;
  /** Strips color codes and small caps and lowercases. */
  cleanText: (text: string) => string;

  /** @deprecated Use `bot.items(query)`. */
  findWindowSlots: (
    filter: { name?: string; lore?: string } | string,
    options?: {
      caseSensitive?: boolean;
      rows?: number[] | null;
      columns?: number[] | null;
      slotRange?: number[] | null;
      window?: { slots: (Item | null)[]; inventoryStart: number };
      toAlpha?: boolean;
    },
  ) => Item[];

  openInventory: () => void;

  onceWithTimeout: <K extends string>(
    event: K,
    timeoutInTicks: number,
  ) => Promise<unknown>;

  getWindowTitle: () => string;
  actions: any;
};
