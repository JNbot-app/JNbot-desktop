cmake_minimum_required(VERSION 3.20)
project(jnbot_needle)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../model.json" NEEDLE_MODEL)
string(JSON NEEDLE_REV GET "${NEEDLE_MODEL}" engineRevision)
set(NODE_INCLUDE_DIR "${NODE_MOBILE_DIR}/android/libnode/include/node")
add_library(node_mobile SHARED IMPORTED)
set_target_properties(node_mobile PROPERTIES IMPORTED_LOCATION "${NODE_MOBILE_DIR}/android/libnode/bin/${ANDROID_ABI}/libnode.so")
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(JNBOT_MOBILE ON)
  set(NEEDLE_OPENBLAS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../engine/mobile/${NEEDLE_REV}/openblas-install" CACHE PATH "" FORCE)
  add_subdirectory(.. inference)
  add_library(jnbot_needle SHARED src/main/cpp/register.cpp)
  target_link_libraries(jnbot_needle jnbot_inference)
else()
  add_library(jnbot_needle SHARED ../binding.cc src/main/cpp/register.cpp)
  target_compile_definitions(jnbot_needle PRIVATE JNBOT_NEEDLE_LINKED JNBOT_NEEDLE_UNSUPPORTED NAPI_VERSION=8)
  target_include_directories(jnbot_needle PRIVATE "${NODE_INCLUDE_DIR}")
endif()
target_link_libraries(jnbot_needle node_mobile log)
target_link_options(jnbot_needle PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
