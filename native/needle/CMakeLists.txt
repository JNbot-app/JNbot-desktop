cmake_minimum_required(VERSION 3.20)
project(jnbot_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(NEEDLE_CPU_SHARED OFF CACHE BOOL "" FORCE)
set(NEEDLE_CPU_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/decoder)
if(JNBOT_MOBILE)
  add_library(jnbot_inference STATIC binding.cc)
  target_compile_definitions(jnbot_inference PRIVATE JNBOT_NEEDLE_LINKED)
else()
  add_library(jnbot_inference MODULE binding.cc)
  set_target_properties(jnbot_inference PROPERTIES PREFIX "" SUFFIX ".node" OUTPUT_NAME "needle")
  if(APPLE)
    target_link_options(jnbot_inference PRIVATE "-undefined" "dynamic_lookup")
  elseif(WIN32)
    target_link_libraries(jnbot_inference PRIVATE "${NODE_LIBRARY}")
  endif()
endif()
target_compile_definitions(jnbot_inference PRIVATE NAPI_VERSION=8)
target_include_directories(jnbot_inference PRIVATE "${NODE_INCLUDE_DIR}" vendor/decoder)
target_link_libraries(jnbot_inference PRIVATE needle_cpu needle_options)
if(ANDROID)
  target_link_options(jnbot_inference PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
